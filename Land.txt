// Cloudinary upload
async function uploadToCloudinary(file) {
    const formData = new FormData();
    formData.append('file', file);
    formData.append('upload_preset', UPLOAD_PRESET);
    const resourceType = file.type.startsWith('video') ? 'video' : 'image';
    const url = `https://api.cloudinary.com/v1_1/${CLOUD_NAME}/${resourceType}/upload`;
    const response = await fetch(url, { method: 'POST', body: formData });
    if (!response.ok) throw new Error('Upload failed');
    const data = await response.json();
    return { url: data.secure_url, type: resourceType, publicId: data.public_id };
}

async function handleMediaSelect(input) {
    const files = Array.from(input.files).slice(0, 6);
    if (!files.length) return;
    const progress = document.getElementById('uploadProgress');
    const progressBar = document.getElementById('progressBar');
    const progressText = document.getElementById('progressText');
    const saveBtn = document.getElementById('saveBtn');
    progress.style.display = 'block';
    saveBtn.disabled = true;
    uploadingCount = files.length;
    let done = 0;
    for (const file of files) {
        try {
            progressText.textContent = `Uploading ${done + 1} of ${files.length}...`;
            progressBar.style.width = `${(done / files.length) * 100}%`;
            const result = await uploadToCloudinary(file);
            pendingMedia.push(result);
            const list = document.getElementById('mediaPreviewList');
            const wrap = document.createElement('div');
            wrap.className = 'media-thumb-wrap';
            if (result.type === 'video') {
                wrap.innerHTML = `<div class="media-thumb-video">🎬</div><button class="media-rm" onclick="removeMedia(this,'${result.url}')">×</button>`;
            } else {
                wrap.innerHTML = `<img class="media-thumb" src="${result.url}"><button class="media-rm" onclick="removeMedia(this,'${result.url}')">×</button>`;
            }
            list.appendChild(wrap);
            done++;
        } catch (e) { alert(`Failed to upload ${file.name}`); }
    }
    progressBar.style.width = '100%';
    progressText.textContent = `✅ ${done} file(s) uploaded`;
    setTimeout(() => { progress.style.display = 'none'; }, 2000);
    saveBtn.disabled = false;
    uploadingCount = 0;
    input.value = '';
}

function removeMedia(btn, url) {
    pendingMedia = pendingMedia.filter(m => m.url !== url);
    btn.parentElement.remove();
}

function handleClickSlide(e) {
    const id = e.currentTarget.id.split('gallery-')[1];
    const p = window.products.find(p=>p.id===id);
    if (!p||!p.media||p.media.length<2) return;
    const rect = e.currentTarget.getBoundingClientRect();
    const dir = e.clientX-rect.left > rect.width/2 ? 1 : -1;
    gIdx[id] = Math.max(0,Math.min(p.media.length-1,(gIdx[id]||0)+dir));
    applySlide(id,p.media.length);
}
function handleTouchStart(e) { const id=e.currentTarget.id.split('gallery-')[1]; tStartX[id]=e.touches[0].clientX; touchActive[id]=true; }
function handleTouchEnd(e) {
    const id=e.currentTarget.id.split('gallery-')[1];
    if(!touchActive[id])return; touchActive[id]=false;
    const diff=tStartX[id]-e.changedTouches[0].clientX;
    if(Math.abs(diff)<40)return;
    const p=window.products.find(p=>p.id===id);
    if(!p||!p.media||p.media.length<2)return;
    gIdx[id]=Math.max(0,Math.min(p.media.length-1,(gIdx[id]||0)+(diff>0?1:-1)));
    applySlide(id,p.media.length);
}
function applySlide(id,total) {
    const s=document.getElementById(`slides-${id}`);
    if(s) s.style.transform=`translateX(-${(gIdx[id]||0)*100}%)`;
    const safeId = id.replace(/[^a-z0-9]/gi,'');
    for(let i=0;i<total;i++){const d=document.getElementById(`dot-${safeId}-${i}`);if(d)d.className='g-dot'+(i===(gIdx[id]||0)?' on':'');}
    const gallery=document.getElementById(`gallery-${id}`);
    if(gallery){const c=gallery.querySelector('.photo-counter');if(c)c.textContent=`${(gIdx[id]||0)+1}/${total}`;}
}

async function saveProductEdit(){
    if(uploadingCount>0){ alert('Please wait for uploads to finish.'); return; }
    const name=document.getElementById('editName').value.trim();
    const brand=document.getElementById('editBrand').value.trim();
    const description=document.getElementById('editDescription').value.trim();
    const price=parseFloat(document.getElementById('editPrice').value)||0;
    const discount=parseFloat(document.getElementById('editDiscount').value)||0;
    const stock=parseInt(document.getElementById('editStock').value)||0;
    const category=document.getElementById('editCategory').value;
    const type=document.getElementById('editType').value;
    const tag=document.getElementById('editTag').value;
    const colors=document.getElementById('editColors').value.split(',').map(c=>c.trim()).filter(Boolean);
    const sizes=getSelectedSizes();
    if(!name||!brand||!price){ alert('Please fill Name, Brand and Price.'); return; }
    document.getElementById('saveBtn').disabled=true;
    document.getElementById('saveBtn').textContent='Saving...';
    try {
        const media=[...pendingMedia];
        const productData = { name,brand,description,price,discount,stock,category,type,tag, sizes:sizes.length?sizes:(type==='shoes'?SHOE_SIZES:CLOTHING_SIZES), colors:colors.length?colors:['Black'], media, available:true };
        if(isAdding){ await window.fbSaveProduct(productData); showNotification('✅ Product added!'); }
        else { await window.fbUpdateProduct(editingId, productData); showNotification('✅ Product updated!'); }
        closeEditModal();
    } catch(e) { alert('Failed to save. Check your internet.'); console.error(e); }
    document.getElementById('saveBtn').disabled=false;
    document.getElementById('saveBtn').textContent='SAVE PRODUCT';
}

function addToCart(id){
    const p=window.products.find(p=>p.id===id); if(!p||!p.available)return;
    const finalPrice=p.discount?(p.price-Math.round(p.price*p.discount/100)):p.price;
    cart.push({id:Date.now(),productId:p.id,name:p.name,brand:p.brand,price:finalPrice,size:document.getElementById(`size-${id}`).value,color:document.getElementById(`color-${id}`).value});
    saveCart(); updateCartUI(); showNotification('✅ Added to cart!');
}
function removeFromCart(i){ cart.splice(i,1); saveCart(); updateCartUI(); }
function updateCartUI(){
    document.getElementById('cartCount').textContent=cart.length;
    const items=document.getElementById('cartItems');
    const empty=document.getElementById('emptyCartMsg');
    const summary=document.getElementById('cartSummary');
    if(!cart.length){items.innerHTML='';empty.style.display='block';summary.style.display='none';return;}
    empty.style.display='none'; summary.style.display='block';
    let baseTotal = cart.reduce((s,i)=>s+i.price,0);
    let discount = 0;
    if(appliedCoupon && appliedCoupon.type==='percent'){ discount = Math.round(baseTotal * appliedCoupon.discount / 100); }
    const finalTotal = baseTotal - discount;
    items.innerHTML=cart.map((item,i)=>`<div class="cart-item"><div><strong>${item.brand} — ${item.name}</strong><br>Size: ${item.size} | Color: ${item.color}<br><span style="color:#00ff00;">GHS ${item.price.toLocaleString()}</span></div><button class="remove-btn" onclick="removeFromCart(${i})">Remove</button></div>`).join('');
    document.getElementById('totalAmount').innerHTML = discount > 0 ? `<span style="text-decoration:line-through;opacity:0.5;font-size:0.85em;">GHS ${baseTotal.toLocaleString()}</span> → GHS ${finalTotal.toLocaleString()}` : `Total: GHS ${finalTotal.toLocaleString()}`;
}
