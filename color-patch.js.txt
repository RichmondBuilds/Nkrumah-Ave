/**
 * NKRUMAH AVE — COLOR PATCH
 * Replaces harsh neon green (#00ff00) with refined deep green (#00CC44)
 * No changes needed to index.html
 */
(function() {

const OLD = '#00ff00';
const NEW = '#00CC44';
const OLD2 = '#00cc00';
const NEW2 = '#00AA33';
const OLD3 = '#00ff44';
const NEW3 = '#00CC55';
const OLD4 = '#0f2a0f';
const NEW4 = '#0A2010';

// ── PATCH ALL STYLESHEETS ────────────────────────────────────────────────────
function patchCSS() {
    // Patch inline styles in <style> tags
    document.querySelectorAll('style').forEach(el => {
        el.textContent = el.textContent
            .replace(/#00ff00/gi, NEW)
            .replace(/#00cc00/gi, NEW2)
            .replace(/#00ff44/gi, NEW3)
            .replace(/#0f2a0f/gi, NEW4);
    });

    // Patch stylesheet rules
    try {
        Array.from(document.styleSheets).forEach(sheet => {
            try {
                Array.from(sheet.cssRules || []).forEach(rule => {
                    if (rule.style) {
                        Array.from(rule.style).forEach(prop => {
                            const val = rule.style.getPropertyValue(prop);
                            if (val.includes('#00ff00') || val.includes('#00cc00') || val.includes('#00ff44')) {
                                rule.style.setProperty(prop,
                                    val.replace(/#00ff00/gi, NEW)
                                       .replace(/#00cc00/gi, NEW2)
                                       .replace(/#00ff44/gi, NEW3)
                                );
                            }
                        });
                    }
                });
            } catch(e) {}
        });
    } catch(e) {}
}

// ── PATCH INLINE STYLES ON ELEMENTS ──────────────────────────────────────────
function patchElements() {
    document.querySelectorAll('[style]').forEach(el => {
        el.setAttribute('style',
            el.getAttribute('style')
                .replace(/#00ff00/gi, NEW)
                .replace(/#00cc00/gi, NEW2)
                .replace(/#00ff44/gi, NEW3)
                .replace(/#0f2a0f/gi, NEW4)
        );
    });
}

// ── WATCH FOR DYNAMICALLY RENDERED ELEMENTS ───────────────────────────────────
const observer = new MutationObserver(() => {
    patchElements();
});
observer.observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['style'] });

// ── OVERRIDE header gradient ──────────────────────────────────────────────────
function patchHeader() {
    const header = document.querySelector('header');
    if (header) {
        header.style.background = `linear-gradient(90deg, ${NEW}, ${NEW2}, ${NEW3})`;
        header.style.boxShadow = `0 4px 20px rgba(0,204,68,0.25)`;
    }
}

// ── PATCH META THEME COLOR ────────────────────────────────────────────────────
const meta = document.querySelector('meta[name="theme-color"]');
if (meta) meta.setAttribute('content', NEW);

// ── RUN ───────────────────────────────────────────────────────────────────────
patchCSS();
patchElements();
patchHeader();

// Also run after DOM loads fully
document.addEventListener('DOMContentLoaded', () => {
    patchCSS();
    patchElements();
    patchHeader();
});

// Run again after products load (Firebase delay)
setTimeout(() => { patchElements(); patchHeader(); }, 1500);
setTimeout(() => { patchElements(); patchHeader(); }, 3000);

console.log('✅ Color patch loaded — neon green replaced with refined green');
})();
